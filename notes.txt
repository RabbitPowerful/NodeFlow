- Every node is a class 2 methods: build() and execute():
    build — draws the node on screen. Runs once when you spawn the node.
    execute — runs the node's logic. Runs every time you hit Run Graph.

class MyValueNode(BaseNode):

    LABEL       = "My Value"   # text on title bar

        super().__init__()
        self._value_id = None
